import { useEffect, useState } from "react";
import { ExternalLink, Camera, Globe2 } from "lucide-react";
import { Link } from "react-router-dom";
import { motion } from "framer-motion";
import DashboardLayout from "../components/DashboardLayout";
import api from "../services/api";

const STATUS_STYLES = {
  draft:     "text-ash bg-ink-700",
  published: "text-signal bg-signal/10",
  expired:   "text-danger bg-danger/10",
};

const NICHE_LABELS = {
  gym_fitness: "Gym & Fitness", salon_spa: "Salon & Spa",
  makeup_studio: "Makeup Studio", real_estate_agency: "Real Estate Agency",
  dental_clinic: "Dental Clinic", construction_company: "Construction Company",
  car_dealership: "Car Dealership", car_rental: "Car Rental",
  hotel_guest_house: "Hotel & Guest House", furniture_interior_design: "Furniture & Interior",
  cleaning_company: "Cleaning Company", bakery_cafe: "Bakery & Café",
  law_firm: "Law Firm", photography_studio: "Photography Studio",
  event_planning: "Event Planning", auto_repair_garage: "Auto Repair & Garage",
};

export default function GeneratedWebsites() {
  const [sites, setSites] = useState([]);
  const [loading, setLoading] = useState(true);
  const [backendError, setBackendError] = useState(false);

  function refresh() {
    api.get("/api/websites")
      .then((res) => { setSites(res.data); setBackendError(false); })
      .catch(() => setBackendError(true))
      .finally(() => setLoading(false));
  }

  useEffect(refresh, []);

  async function handleScreenshot(id) {
    try { await api.post(`/api/websites/${id}/screenshot`); refresh(); }
    catch { /* non-fatal */ }
  }

  async function handleStatus(id, status) {
    try { await api.patch(`/api/websites/${id}/status`, null, { params: { status } }); refresh(); }
    catch { /* non-fatal */ }
  }

  return (
    <DashboardLayout>
      <h1 className="font-display font-semibold text-2xl text-ash-light mb-1">Generated Websites</h1>
      <p className="text-ash text-sm mb-8">
        Demo sites generated for your leads — each one ready to preview and share.
      </p>

      {backendError && (
        <div className="card p-4 mb-6 border border-danger/30 text-sm text-ash">
          Can't reach the backend right now — check that it's deployed and reachable.
        </div>
      )}

      {loading ? (
        <div className="text-ash text-sm">Loading…</div>
      ) : sites.length === 0 ? (
        <div className="card p-12 text-center">
          <Globe2 size={32} className="mx-auto mb-4 text-ash/20" />
          <p className="text-ash text-sm mb-2">No websites generated yet.</p>
          <p className="text-ash/50 text-xs">
            Go to Lead Finder, find or add a lead, then click "Generate site".
          </p>
        </div>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
          {sites.map((site, i) => (
            <motion.div
              key={site.id}
              initial={{ opacity: 0, y: 8 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ delay: i * 0.05 }}
              className="card overflow-hidden"
            >
              {/* Preview area */}
              <div className="h-36 bg-ink-700 flex items-center justify-center relative overflow-hidden">
                {site.screenshot_path ? (
                  <img src={site.screenshot_path} alt={site.business_name}
                    className="w-full h-full object-cover" />
                ) : (
                  <div className="text-center">
                    <Globe2 size={24} className="mx-auto mb-2 text-ash/20" />
                    <span className="text-ash/40 text-xs">No screenshot yet</span>
                  </div>
                )}
              </div>

              <div className="p-4">
                <p className="text-ash-light font-semibold text-sm mb-0.5">{site.business_name}</p>
                <p className="text-ash/50 text-xs mb-3">
                  {NICHE_LABELS[site.template_key] || site.template_key}
                </p>

                <div className="flex items-center justify-between mb-4">
                  <span className={`text-xs px-2 py-0.5 rounded-full ${STATUS_STYLES[site.status] || STATUS_STYLES.draft}`}>
                    {site.status}
                  </span>
                  <select
                    className="bg-transparent text-xs text-ash border border-ink-600 rounded px-1.5 py-0.5"
                    value={site.status}
                    onChange={(e) => handleStatus(site.id, e.target.value)}
                  >
                    <option value="draft">draft</option>
                    <option value="published">published</option>
                    <option value="expired">expired</option>
                  </select>
                </div>

                <div className="flex gap-2">
                  {/* Fixed: use demo_token not site.id */}
                  <Link
                    to={`/demo/${site.demo_token}`}
                    target="_blank"
                    className="btn-secondary text-xs flex items-center gap-1 flex-1 justify-center"
                  >
                    <ExternalLink size={12} /> Preview
                  </Link>
                  <button
                    onClick={() => handleScreenshot(site.id)}
                    className="btn-secondary text-xs flex items-center gap-1 px-2"
                    title="Capture screenshot"
                  >
                    <Camera size={12} />
                  </button>
                </div>
              </div>
            </motion.div>
          ))}
        </div>
      )}
    </DashboardLayout>
  );
}
